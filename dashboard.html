// ============================================================ // BUY RO - DENGAN MODAL SIZE & WARNA // ============================================================ function buyRO(productId, price) { const user = getCurrentUser(); if (!user) { showToast('Silakan login!', 'error'); return; } if (user.status !== 'verified') { showToast('❌ Anda harus terverifikasi (verified) untuk melakukan repeat order.', 'error'); return; } const products = loadProducts(); const product = products.find(p => p.id === productId); if (!product) { showToast('Produk tidak ditemukan', 'error'); return; } if (product.stock <= 0) { showToast('Stok habis', 'error'); return; } // Buat opsi size let sizeOptions = ''; const sizes = product.sizes || ['28', '29', '30', '31', '32', '33', '34', '36', '38', '40']; sizes.forEach(s => { sizeOptions += ``; }); // Buat opsi warna let colorOptions = ''; const colors = product.colors || ['Indigo Blue Raw Denim', 'Black Stretch', 'Light Wash', 'Dark Navy', 'Grey Stone', 'White']; colors.forEach(c => { colorOptions += ``; }); // Tampilkan modal let html = `

Beli Produk (RO)

Pilih size dan warna untuk ${product.name}

${product.image && (product.image.startsWith('data:image') || product.image.startsWith('http')) ? `
` : `
` }
${product.name}
Harga: Rp ${price.toLocaleString()}
Stok: ${product.stock}
${product.tag || 'PRODUK'}
Deskripsi${product.desc || '-'}
Nomor Seri${product.serial || '-'}

💳 Transfer ke Rekening Admin:

${loadBankInfo().bank} - ${loadBankInfo().account} a.n. ${loadBankInfo().name}

Upload bukti transfer
`; openModal(html); // File input handler const fileInput = document.getElementById('roProofFile'); const preview = document.getElementById('roProofPreview'); const label = document.getElementById('roProofLabel'); fileInput.addEventListener('change', function(e) { const file = e.target.files[0]; if (file) { const reader = new FileReader(); reader.onload = function(event) { preview.innerHTML = ``; preview.classList.add('show'); label.textContent = file.name; fileInput.dataset.imageData = event.target.result; }; reader.readAsDataURL(file); } }); document.getElementById('confirmROBuy').addEventListener('click', function() { const size = document.getElementById('roSize').value; const color = document.getElementById('roColor').value; const qty = parseInt(document.getElementById('roQty').value) || 1; const proofData = fileInput.dataset.imageData; if (qty <= 0) { showToast('Jumlah minimal 1!', 'error'); return; } if (qty > product.stock) { showToast('Stok tidak mencukupi!', 'error'); return; } if (!proofData) { showToast('Upload bukti transfer terlebih dahulu!', 'error'); return; } let products = loadProducts(); const prod = products.find(p => p.id === productId); if (!prod || prod.stock < qty) { showToast('Stok habis!', 'error'); closeModal(); return; } prod.stock -= qty; saveProducts(products); let users = loadUsers(); const current = users.find(u => u.id === user.id); if (!current) return; const totalPrice = price * qty; if (!current.transactions) current.transactions = []; const bankInfo = loadBankInfo(); current.transactions.push({ type: 'ro_purchase', productId: productId, productName: product.name, size: size, color: color, qty: qty, amount: totalPrice, date: new Date().toISOString(), status: 'pending', note: `Menunggu pembayaran transfer ke ${bankInfo.bank} - ${bankInfo.account}`, proof: proofData }); if (!current.purchase_history) current.purchase_history = []; current.purchase_history.push({ productId: productId, productName: product.name, size: size, color: color, qty: qty, price: totalPrice, date: new Date().toISOString(), status: 'pending' }); saveUsers(users); // Buat order const orders = loadOrders(); orders.push({ id: getNextId(orders), order_number: 'RO-' + Date.now().toString().slice(-6), user_id: user.id, user_name: user.name || user.username, items: [{ product_id: product.id, name: product.name, price: price, qty: qty, size: size, color: color }], total: totalPrice, status: 'pending', type: 'ro', proof_image: proofData, createdAt: new Date().toISOString(), updatedAt: new Date().toISOString() }); saveOrders(orders); closeModal(); showToast(`✅ Pesanan RO ${product.name} (${size} · ${color}) x${qty} berhasil! Silakan transfer ke rekening admin.`, 'success'); renderAll(); showSection('history'); }); document.getElementById('cancelROBuy').addEventListener('click', function() { closeModal(); }); }
